//@version=5
strategy("Double Bollinger Band (DBB) Momentum", overlay=true, initial_capital=10000, default_qty_type=strategy.fixed, default_qty_value=10)

// --- Inputs ---
length       = input.int(20, "BB Length")
src          = input(close, "Source")
mult1        = input.float(1.0, "Inner Mult (1SD)")
mult2        = input.float(2.0, "Outer Mult (2SD)")
minHoldBars  = input.int(5, "Min Hold Bars (before exit allowed)", minval=1)
expandThresh  = input.float(1.02, "Band Expansion Threshold", minval=1.0, step=0.01)
lastEntryHour = input.int(22, "Last Entry Hour (Broker Time)", minval=0, maxval=23)

// --- No New Entries After 10PM ---
brokerHour = hour(time)
entryWindowOpen = brokerHour < lastEntryHour

// --- Calculations ---
basis = ta.sma(src, length)
dev1  = mult1 * ta.stdev(src, length)
dev2  = mult2 * ta.stdev(src, length)

// Upper Bands
u2 = basis + dev2
u1 = basis + dev1
// Lower Bands
l1 = basis - dev1
l2 = basis - dev2

// --- Band Expansion Filter ---
bbWidth   = u1 - l1
expanding = bbWidth > bbWidth[1] * expandThresh

// --- Min Hold Tracker ---
var int barsSinceLongEntry  = 0
var int barsSinceShortEntry = 0

if strategy.position_size > 0
    barsSinceLongEntry += 1
else
    barsSinceLongEntry := 0

if strategy.position_size < 0
    barsSinceShortEntry += 1
else
    barsSinceShortEntry := 0

// --- Visuals ---
plot(basis, "Midline", color=color.gray)
p_u2 = plot(u2, "Upper 2", color=color.green)
p_u1 = plot(u1, "Upper 1", color=color.green)
p_l1 = plot(l1, "Lower 1", color=color.red)
p_l2 = plot(l2, "Lower 2", color=color.red)

// Highlight bars where bands are expanding
bgcolor(expanding ? color.new(color.blue, 92) : na, title="Expanding")

// Fill Zones
fill(p_u1, p_u2, color=color.new(color.green, 90), title="Bullish Momentum")
fill(p_l1, p_l2, color=color.new(color.red, 90), title="Bearish Momentum")

// --- Shared Signals ---
longEntrySignal  = ta.crossover(close, u1) and expanding and entryWindowOpen
shortEntrySignal = ta.crossunder(close, l1) and expanding and entryWindowOpen
longExitSignal = ta.crossunder(close, u1) and barsSinceLongEntry >= minHoldBars
shortExitSignal = ta.crossover(close, l1) and barsSinceShortEntry >= minHoldBars

// --- Entry Logic (autonomous from plotted signals) ---
if longEntrySignal
    strategy.entry("DBB Long", strategy.long)

if shortEntrySignal
    strategy.entry("DBB Short", strategy.short)

// --- Exit Logic (autonomous from plotted signals) ---
if longExitSignal
    strategy.close("DBB Long")
    barsSinceLongEntry := 0

if shortExitSignal
    strategy.close("DBB Short")
    barsSinceShortEntry := 0

// --- Plot Markers ---
plotshape(longEntrySignal, title="Long Entry Plot", style=shape.labelup, location=location.belowbar, color=color.new(color.lime, 0), text="BUY", textcolor=color.white, size=size.tiny)
plotshape(shortEntrySignal, title="Short Entry Plot", style=shape.labeldown, location=location.abovebar, color=color.new(color.red, 0), text="SELL", textcolor=color.white, size=size.tiny)
plotshape(longExitSignal, title="Long Exit Plot", style=shape.circle, location=location.abovebar, color=color.new(color.aqua, 0), text="X", textcolor=color.white, size=size.tiny)
plotshape(shortExitSignal, title="Short Exit Plot", style=shape.circle, location=location.belowbar, color=color.new(color.orange, 0), text="X", textcolor=color.white, size=size.tiny)

// --- Webhook Alert Conditions ---
alertcondition(longEntrySignal,  "DBB Long Entry",  '{"action":"buy","symbol":"{{ticker}}","price":{{close}},"tf":"{{interval}}"}')
alertcondition(longExitSignal,   "DBB Long Exit",   '{"action":"close_buy","symbol":"{{ticker}}","price":{{close}},"tf":"{{interval}}"}')
alertcondition(shortEntrySignal, "DBB Short Entry", '{"action":"sell","symbol":"{{ticker}}","price":{{close}},"tf":"{{interval}}"}')
alertcondition(shortExitSignal,  "DBB Short Exit",  '{"action":"close_sell","symbol":"{{ticker}}","price":{{close}},"tf":"{{interval}}"}')
